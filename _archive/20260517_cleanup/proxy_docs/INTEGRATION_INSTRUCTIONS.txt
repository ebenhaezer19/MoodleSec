"""
EXAMPLE: If you have a /api/scan endpoint like:

@app.post("/api/scan")
async def start_scan(request: ScanRequest):
    ...
    return {"status": "started", "scan_id": scan_id}


MODIFY IT TO:

@app.post("/api/scan")
async def start_scan(request: ScanRequest):
    scan_id = str(uuid.uuid4())
    
    # Log scan start
    if debug_logger:
        debug_logger.log_scan_start(
            scan_id=scan_id,
            scan_type='SCAN_NOW',
            target_url=request.target_url
        )
    
    try:
        # Reinitialize scanner with fresh payloads
        scanner_engine.initialize_scanners()
        
        # Perform scan
        results = scanner_engine.scan(
            url=request.target_url,
            method=request.method or 'GET',
            params=request.params,
            request_body=request.body or "",
            response_body=request.response_body or "",
            request_headers=request.headers or {},
            response_headers=request.response_headers or {},
            status_code=request.status_code or 200
        )
        
        # Log scan complete
        if debug_logger:
            debug_logger.log_scan_complete(
                scan_id=scan_id,
                findings_count=len(results.get('findings', [])),
                status='SUCCESS'
            )
        
        return {
            "status": "completed",
            "scan_id": scan_id,
            "results": results
        }
    
    except Exception as e:
        # Log scan failure
        if debug_logger:
            debug_logger.log_scan_complete(
                scan_id=scan_id,
                findings_count=0,
                status='FAILED'
            )
        
        return {
            "status": "error",
            "scan_id": scan_id,
            "error": str(e)
        }
"""

"""
MODIFY the ScannerEngine.__init__ to accept debug_logger:

class ScannerEngine:
    def __init__(self, debug_logger=None):
        self.debug_logger = debug_logger
        
        # Initialize detectors...
        self.sql_detector = SQLInjectionDetector()
        self.xss_detector = XSSDetector()
        # ... rest of init ...
        
        # Log payload loading
        if self.debug_logger:
            self._log_payload_initialization()
    
    def _log_payload_initialization(self):
        '''Log all loaded payloads at startup'''
        try:
            if self.sql_detector.payload_repo:
                sql_payloads = self.sql_detector.payload_repo.get_top_payloads("SQL Injection", limit=100)
                if sql_payloads:
                    self.debug_logger.log_payload_loaded(
                        category='SQL_INJECTION',
                        count=len(sql_payloads),
                        payload_list=[p.get('payload_text', '')[:50] for p in sql_payloads]  # First 50 chars
                    )
            
            if self.xss_detector.payload_repo:
                xss_payloads = self.xss_detector.payload_repo.get_top_payloads("XSS", limit=100)
                if xss_payloads:
                    self.debug_logger.log_payload_loaded(
                        category='XSS',
                        count=len(xss_payloads),
                        payload_list=[p.get('payload_text', '')[:50] for p in xss_payloads]
                    )
            
            if self.csrf_validator.payload_repo:
                csrf_payloads = self.csrf_validator.payload_repo.get_top_payloads("CSRF", limit=100)
                if csrf_payloads:
                    self.debug_logger.log_payload_loaded(
                        category='CSRF',
                        count=len(csrf_payloads),
                        payload_list=[p.get('payload_text', '')[:50] for p in csrf_payloads]
                    )
        except Exception as e:
            print(f"[!] Error logging payload initialization: {e}")
"""
